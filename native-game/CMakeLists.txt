# Standalone build: no inclusion of game/native, engine SDK or generated bindings.
# source: SDL2/README-cmake; CMake FindPNG, FindFreetype and FindPkgConfig manuals.

add_executable(Transartica app/main.cpp app/session_ui.cpp app/session_ui_panels.cpp app/session_ui_launcher.cpp)
target_link_libraries(Transartica PRIVATE transartica_view)
target_compile_definitions(Transartica PRIVATE SDL_MAIN_HANDLED)
if(APPLE)
  # install_name_tool(1): reserve load-command space for relocatable dylib paths.
  target_link_options(Transartica PRIVATE -Wl,-headerpad_max_install_names)
endif()
if(MSVC)
  target_compile_options(transartica_core PRIVATE /W4)
  target_compile_options(Transartica PRIVATE /W4)
else()
  target_compile_options(transartica_core PRIVATE -Wall -Wextra -Wpedantic)
  target_compile_options(Transartica PRIVATE -Wall -Wextra -Wpedantic)
endif()
# CMake custom targets run on each build, so changed manifest dependencies are
# revalidated even without C++ changes. The stager skips byte-identical copies.
set(TA_RUNTIME_RESOURCES "${CMAKE_CURRENT_BINARY_DIR}/resources")
add_custom_target(transartica_resources ALL
  COMMAND "${Python3_EXECUTABLE}" -B "${CMAKE_CURRENT_SOURCE_DIR}/packaging/stage_runtime.py"
    --repo "${CMAKE_CURRENT_SOURCE_DIR}/.." --output "${TA_RUNTIME_RESOURCES}"
  BYPRODUCTS "${TA_RUNTIME_RESOURCES}/resource-inventory.json"
  COMMENT "Staging Transartica runtime resources"
  VERBATIM)
add_dependencies(Transartica transartica_resources)
# Multi-configuration generators place executables in a configuration directory.
set_target_properties(Transartica PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/$<0:>")

# Keep the install tool in the build tree so installation does not read source assets.
set(TA_INSTALL_TOOL_DIR "${CMAKE_CURRENT_BINARY_DIR}/CMakeFiles/resource-install")
foreach(install_tool stage_runtime.py macos_bundle.py macos_bundle_macho.py)
  configure_file("packaging/${install_tool}" "${TA_INSTALL_TOOL_DIR}/${install_tool}" COPYONLY)
endforeach()
configure_file(packaging/install_resources.cmake.in install_resources.cmake @ONLY)
install(SCRIPT "${CMAKE_CURRENT_BINARY_DIR}/install_resources.cmake")
